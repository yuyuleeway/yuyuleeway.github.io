function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var params = e.parameter;
  
  sheet.appendRow([
    new Date(),
    params.name,
    params.phone,
    params.lineId,
    params.service,
    params.removal,
    params.date,
    params.time,
    params.note
  ]);
  
  return ContentService.createTextOutput(
    JSON.stringify({ "result": "success" })
  ).setMimeType(ContentService.MimeType.JSON);
}
